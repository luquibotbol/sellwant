# Content signals (https://contentsignals.org). These say what may be done
# with what is collected, and are deliberately different answers:
#
#   search=yes    index the listings; being findable is the point.
#   ai-input=yes  ground live answers on them. A model that can answer
#                 "who's selling a ticket to X" is doing our distribution.
#   ai-train=no   listings are other people's plans and prices, posted to
#                 trade this weekend, not to become training data. Reading
#                 them now is useful to us; keeping them forever is not, and
#                 it is not ours to grant.
Content-Signal: search=yes, ai-input=yes, ai-train=no
