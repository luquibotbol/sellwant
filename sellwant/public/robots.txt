User-agent: *
Allow: /
Allow: /api/listings.json
Allow: /terms
Allow: /privacy

# Private surfaces. They already require a session and redirect when signed
# out, but naming them keeps them out of indexes and out of crawl budget
# rather than relying on a redirect to imply it.
Disallow: /deal/
Disallow: /deals
Disallow: /u/
Disallow: /offers
Disallow: /my-listings
Disallow: /profile
Disallow: /onboarding
Disallow: /auth/

# Sign-in is a dead end for a crawler -- no content, and it is not the page
# anyone should arrive on from a search.
Disallow: /signin

# Content signals (https://contentsignals.org). Deliberately different
# answers to three different questions:
#
#   search=yes    index the listings. Being findable is the entire point.
#   ai-input=yes  ground live answers on them. A model that can answer
#                 "who is selling a ticket to X this weekend" is doing our
#                 distribution for us, and the data is public either way.
#   ai-train=no   listings are other people's plans and prices, posted to
#                 trade this weekend and then irrelevant. Reading them now
#                 helps everyone; keeping them permanently in a model helps
#                 nobody here, and it is not ours to grant on their behalf.
Content-Signal: search=yes, ai-input=yes, ai-train=no
